\begin{frame}{What Is Implemented Now}
\textbf{Pipeline status:}
\begin{itemize}
  \item Counterfactual categories: \texttt{goal\_move}, \texttt{start\_goal\_swap}, \texttt{rotate\_90}, \texttt{reflect\_vertical}, \texttt{transpose}
  \item Default generation: \textbf{10 pairs per category} (50 total)
  \item Manifests include \texttt{category}, \texttt{goal\_a}, \texttt{goal\_b}
  \item Preflight validates category-specific constraints before expensive runs
  \item Evaluator outputs separate MLP vs Linear belief-action metrics
\end{itemize}

\vspace{0.4em}
\textbf{Interpretability improvement:}
\begin{itemize}
  \item Invalid and disruptive cases are excluded from 2x2 belief-action tables
  \item Invalid/disruptive cases are still reported explicitly for diagnosis
\end{itemize}
\end{frame}

\begin{frame}{Key Artifacts and Metric Semantics}
\textbf{Core artifacts:}
\begin{itemize}
  \item Pair manifest: \texttt{data/cf/pair\_manifest.json}
  \item Per-pair traces: \texttt{A.json}, \texttt{B.json}, \texttt{patched.json}
  \item Eval manifest: \texttt{manifest\_for\_counterfactual\_activation\_patching.json}
  \item Outputs: \texttt{per\_pair\_results.(jsonl,csv)}, \texttt{aggregate\_summary.json}, \texttt{report.md}
\end{itemize}

\vspace{0.4em}
\textbf{What the action metrics mean:}
\begin{itemize}
  \item Read every metric as \textbf{(behavior being scored, reference used for scoring)}
  \item \texttt{A\_target} (alias \texttt{A\_new}): \textbf{(same intervened trace, true grid-B optimal policy)}
  \item \texttt{A\_base} (alias \texttt{A\_orig}): \textbf{(same intervened trace, true grid-A optimal policy)}
  \item These are scored on the \textbf{actual grids A and B}, not on the decoded map
  \item Non-integers are expected: each value is a ratio \texttt{aligned\_steps / evaluated\_steps}
\end{itemize}

\vspace{0.4em}
\textbf{Belief vs action distinction:}
\begin{itemize}
  \item Belief readout uses the \textbf{decoded cognitive map / decoded goal from probes}
  \item Decoded-map information is used for \textbf{belief readout only}, not for \texttt{A\_target} / \texttt{A\_base}
  \item Paper comparison: \texttt{Acc. GT} in Table 2 is \textbf{(original unpatched behavior, ground-truth grid)}, so it is not directly equivalent to \texttt{A\_base}
\end{itemize}

\vspace{0.4em}
\textbf{Disruptive reason now reported per pair:}
\begin{itemize}
  \item Triggered when \texttt{A\_target < 0.35} and \texttt{A\_base < 0.35}
  \item Report includes threshold and actual values for interpretability
\end{itemize}
\end{frame}

\begin{frame}{How to Read the Belief--Action 2x2}
\textbf{Definitions used in the table:}
\begin{itemize}
  \item \textbf{Belief F}: the probe-decoded cognitive map does \textit{not} align with the target counterfactual grid in the prompt.
  \item \textbf{Action F}: the same intervened trace does not satisfy Action=True, i.e., it fails \($A_{\text{target}} > A_{\text{base}}$ and $A_{\text{target}} \ge 0.70$\), after excluding disruptive pairs.
\end{itemize}

\vspace{0.3em}
\textbf{Important distinction:}
\begin{itemize}
  \item Belief uses the \textbf{decoded map}; action uses optimality on the \textbf{actual grids A and B}
  \item For Mario: there is only one intervened trace; ``base'' and ``target'' mean scoring that same trace against true grid A or true grid B, respectively
\end{itemize}

\vspace{0.3em}
\textbf{Cell interpretation for this run:}
\begin{itemize}
  \item \textbf{Belief T, Action T} (count = 2 in MLP): belief aligns with target and behavior transfers; action matches belief.
  \item \textbf{Belief T, Action F}: belief aligns with target but behavior does not transfer; this is the belief--action gap.
  \item \textbf{Belief F, Action T}: present only for Linear here; interpreted as probable probe decoding mismatch (e.g., belief encoded non-linearly).
  \item \textbf{Belief F, Action F}: neither decoded belief nor behavior transfers to the target.
\end{itemize}
\end{frame}

\begin{frame}{Expansion Readout (Current Run)}
\textbf{Setup:}
\begin{itemize}
  \item Layer sweep: coarse \{9,12,15,18,21\} + local refine around best
  \item Action-threshold sensitivity: \{0.65, 0.70, 0.75\}, disruptive threshold fixed at 0.35
  \item Early-stop disabled for comparability across configs
\end{itemize}

\vspace{0.3em}
\textbf{Result summary:}
\begin{itemize}
  \item At threshold 0.70, tested layers are effectively tied (\texttt{action\_true\_rate}=0.25, \texttt{disruptive\_rate}=0.50).
  \item Selected ``best'' layer is 7 by ranking tie-break; no clear single-layer advantage in this tested range.
  \item Conclusion: current effect is not strongly layer-sensitive under these thresholds.
\end{itemize}
